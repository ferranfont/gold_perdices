//@version=5
indicator("Simple RSI", overlay=false)

// --- Inputs ---
length = input.int(14, "RSI Period", minval=1)
useSmoothing = input.bool(false, "Enable Smoothing (EMA)?")
smoothPeriod = input.int(5, "Smoothing Period")
useHTF = input.bool(false, "Use Higher Timeframe?")
htf = input.timeframe("60", "Higher Timeframe Resolution")
overbought = input.float(70.0, "Overbought Level")
oversold = input.float(30.0, "Oversold Level")

// --- Calculation ---
// 1. Calculate base RSI
rsiValue = ta.rsi(close, length)

// 2. Apply Smoothing if requested
rsiProcessed = useSmoothing ? ta.ema(rsiValue, smoothPeriod) : rsiValue

// 3. Fetch Higher Timeframe data if requested
// requesting the processed RSI from the specific timeframe
rsiFinal = useHTF ? request.security(syminfo.tickerid, htf, rsiProcessed) : rsiProcessed

// --- Plotting ---
plot(rsiFinal, "RSI", color=color.purple, linewidth=2)

// Levels
h1 = hline(overbought, "Overbought", color=color.red, linestyle=hline.style_dashed)
h2 = hline(oversold, "Oversold", color=color.green, linestyle=hline.style_dashed)
fill(h1, h2, color=color.new(color.purple, 90), title="Middle Zone")

// Visual Highlights
bgcolor(rsiFinal >= overbought ? color.new(color.red, 90) : na, title="Overbought Background")
bgcolor(rsiFinal <= oversold ? color.new(color.green, 90) : na, title="Oversold Background")
